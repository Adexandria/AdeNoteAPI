29 public async Task<ActionResult> Handle(TranslatePageRequest request, CancellationToken cancellationToken)
30 {
31 var currentBook = cacheService.Get<Book>($"{_bookCacheKey}:{request.UserId}:{request.BookId}")
32 ?? await bookRepository.GetAsync(request.BookId, request.UserId);
33
34 if (currentBook == null)
35 return ActionResult<TranslationDto>.Failed("Book doesn't exist", (int)HttpStatusCode.NotFound);
36
37 var currentBookPage = cacheService.Get<Page>($"{_pageCacheKey}:{request.BookId}:{request.PageId}")
38 ?? await pageRepository.GetBookPage(request.BookId, request.PageId);
39
40 if (currentBookPage == null)
41 return ActionResult<TranslationDto>.Failed("page doesn't exist", (int)HttpStatusCode.NotFound);
42
43 var existingTranslatedPage = cacheService.Get<TranslationDto>($"{_pageCacheKey}:{request.BookId}:{request.PageId}:{request.TranslatedLanguage.ToLower()}");
44
45 if (existingTranslatedPage != null)
46 {
47 return ActionResult<TranslationDto>.SuccessfulOperation(existingTranslatedPage);
48 }
49
50 var translationLanguages = cacheService.Get<Dictionary<string, string>>("translation_languages");
51
52 var transliterationLanguages = cacheService.Get<Dictionary<string, string>>("transliteration_languages");
53
54 if (translationLanguages == null)
55 {
56 var languagesResponse = textTranslation.GetSupportedLanguages("translation", "transliteration", cancellationToken: cancellationToken);
57 if (languagesResponse.NotSuccessful)
58 {
59 return ActionResult<TranslationDto>
60 .Failed("Failed to fetch languages", StatusCodes.Status400BadRequest);
61 }
62 cacheService.Set("translation_languages", languagesResponse.Data.TranslationLanguages);
63 cacheService.Set("transliteration_languages", languagesResponse.Data.TransliterationLanguages);
64 cacheService.Set("etag", languagesResponse.Data.ETag);
65
66 translationLanguages = languagesResponse.Data.TranslationLanguages;
67 transliterationLanguages = languagesResponse.Data.TransliterationLanguages;
68 }
69
70 var translationLanguage = translationLanguages.FirstOrDefault(s => s.Key.ToUpper() == request.TranslatedLanguage.ToUpper()).Value;
71
72 if (translationLanguage == null)
73 {
74 return ActionResult<TranslationDto>
75 .Failed("The language is not supported", StatusCodes.Status400BadRequest);
76 }
77
78 var translatedResponse = await textTranslation.TranslatePage(currentBookPage.Content, translationLanguage, cancellationToken);
79
80 if (translatedResponse.NotSuccessful)
81 {
82 return ActionResult<TranslationDto>
83 .Failed(translatedResponse.Errors.FirstOrDefault(), StatusCodes.Status400BadRequest);
84 }
85
86 request.TranslatedLanguage = translationLanguages.FirstOrDefault(s => s.Key.ToUpper() == request.TranslatedLanguage.ToUpper()).Key;
87
88 var detectedLanguage = translationLanguages.FirstOrDefault(s => s.Value.ToUpper() == translatedResponse.Data[1].ToUpper()).Key ?? translatedResponse.Data[1];
89
90 var transliterationLanguage = transliterationLanguages.FirstOrDefault(s => s.Value.ToUpper() == translationLanguage.ToUpper()).Key;
91 TranslationDto translatedPage;
92
93 if (transliterationLanguage != null)
94 {
95 var transliterationResponse = await textTranslation.TransliteratePage(translatedResponse.Data[0], translationLanguage, transliterationLanguage, cancellationToken);
96
97 if (transliterationResponse.NotSuccessful)
98 {
99 return ActionResult<TranslationDto>
100 .Failed(transliterationResponse.Errors.FirstOrDefault(), StatusCodes.Status400BadRequest);
101 }
102
103 translatedPage = new TranslationDto(currentBookPage.Content,
104 translatedResponse.Data[0],
105 detectedLanguage, request.TranslatedLanguage, transliterationResponse.Data);
106
107 cacheService.Set($"{_pageCacheKey}:{request.BookId}:{request.PageId}:{request.TranslatedLanguage.ToLower()}", translatedPage, DateTime.UtcNow.AddMinutes(10));
108
109 return ActionResult<TranslationDto>.SuccessfulOperation(translatedPage);
110 }
111 translatedPage = new TranslationDto(currentBookPage.Content,
112 translatedResponse.Data[0],
113 detectedLanguage, request.TranslatedLanguage);
114
115 cacheService.Set($"{_pageCacheKey}:{request.BookId}:{request.PageId}:{request.TranslatedLanguage.ToLower()}", translatedPage, DateTime.UtcNow.AddMinutes(10));
116
117 return ActionResult<TranslationDto>.SuccessfulOperation(translatedPage);
118 }

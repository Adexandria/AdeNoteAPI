26 ?? await bookRepository.GetAsync(request.BookId, request.UserId);
27
28 if (currentBook == null)
29 return ActionResult.Failed("Book doesn't exist", (int)HttpStatusCode.NotFound);
30
31 var currentBookPage = cacheService.Get<Page>($"{_pageCacheKey}:{request.BookId}:{request.PageId}")
32 ?? await pageRepository.GetBookPage(request.BookId, request.PageId, true);
33 if (currentBookPage == null)
34 return ActionResult.Failed("page doesn't exist", (int)HttpStatusCode.NotFound);
35
36 var page = request.UpdatePage.Map<PageUpdateDTO, Page>();
37
38 page.Id = request.PageId;
39 page.BookId = request.BookId;
40
41 page.SetModifiedDate();
42
43 var commitStatus = await pageRepository.Update(page, currentBookPage);
44 if (!commitStatus)
45 return ActionResult.Failed("Failed to update page");
46
47 cacheService.Set($"{_pageCacheKey}:{request.BookId}:{request.PageId}", page, DateTime.UtcNow.AddMinutes(30));
48
49 return ActionResult.SuccessfulOperation();

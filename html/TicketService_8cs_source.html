194 public async Task<ActionResult> UpdateTicket(string status, Guid adminId, Guid ticketId, SolvedTicketDto? solvedTicketDto)
195 {
196 if (ticketId == Guid.Empty)
197 return ActionResult.Failed("Invalid ticket id", StatusCodes.Status400BadRequest);
198
199 var currentTicket = await ticketRepository.GetTicket(ticketId);
200 if (currentTicket == null)
201 return ActionResult.Failed("There are no existing ticket", StatusCodes.Status400BadRequest);
202
203 var isStatus = Enum.TryParse(status, out Status newStatus);
204
205 if (!isStatus)
206 {
207 return ActionResult.Failed("Invalid status", StatusCodes.Status400BadRequest);
208 }
209
210 currentTicket.Status = newStatus;
211
212 currentTicket.AdminId = adminId;
213
214 currentTicket.SetModifiedDate();
215
216 var commitStatus = await ticketRepository.Update(currentTicket);
217
218 if (!commitStatus)
219 {
220 return ActionResult.Failed("Failed to update ticket", StatusCodes.Status400BadRequest);
221 }
222
223 var admin = await userRepository.GetUser(adminId);
224
225
226 string message = JsonSerializer.Serialize(new
227 {
228 status = newStatus.GetDescription(),
229 firstName = currentTicket.User.FirstName,
230 lastName = currentTicket.User.LastName,
231 ticketId = $"tk{currentTicket.Id.ToString("N")[^5..]}",
232 emailAddress = currentTicket.User.Email,
233 issue = currentTicket.Issue,
234 lastUpdated = currentTicket.Modified.ToShortDateString(),
235 dateSubmitted = currentTicket.Created.ToShortDateString(),
236 agent = admin.FirstName + " " + admin.LastName,
237 resolvedDetails = solvedTicketDto?.ResolutionDetails
238 });
239
240 messagingService.Publish(message, eventConfiguration.Exchange, eventConfiguration.RoutingKey);

242 return ActionResult.SuccessfulOperation();

136 public async Task<IActionResult> CreatePage([ValidGuid("Invalid book id")] Guid bookId, [FromForm]PageCreateDTO pageCreate)
137 {
138 var memoryStream = new MemoryStream();
139
140 if (pageCreate.File != null)
141 {
142 if (pageCreate.File?.ContentType != MimeType.mp4.GetDescription())
143 return BadRequest("Incorrect video type");
144
145 await pageCreate?.File?.CopyToAsync(memoryStream);
146
147 memoryStream.Position = 0;
148 }
149
150 var response = await Application.SendAsync(new CreatePageRequest()
151 {
152 BookId = bookId,
153 Content = pageCreate.Content,
154 Title = pageCreate.Title,
155 UserId = CurrentUser,
156 Description = pageCreate.Description,
157 Stream = memoryStream
158 });
159 return response.Response();
160 }

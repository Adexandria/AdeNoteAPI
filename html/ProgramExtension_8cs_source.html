154 public static void RegisterAuthorization(this IServiceCollection serviceCollection)
155 {
156 serviceCollection.AddAuthorization(options =>
157 {
158 options.AddPolicy("User", policy =>
159 {
160 policy.AddAuthenticationSchemes(JwtBearerDefaults.AuthenticationScheme);
161 policy.Requirements.Add(new RoleRequirement(Role.User));
162 });
163
164 options.AddPolicy("Owner", policy =>
165 {
166 policy.AddAuthenticationSchemes(JwtBearerDefaults.AuthenticationScheme);
167 policy.AddRequirements(new RoleRequirement(Role.Admin, Role.SuperAdmin));
168 });
169
170 options.AddPolicy("sso", new AuthorizationPolicyBuilder("SSO")
171 .RequireAuthenticatedUser().Build());
172 });

175 public static void RegisterValidation(this IServiceCollection services)
176 {
177 var validationTypes = Assembly.GetExecutingAssembly().GetTypes()
178 .Where(s => s.GetInterfaces()
179 .Any(p=>p.Name == typeof(IValidator<>).Name) && !s.IsAbstract);
180
181
182 validationTypes.Foreach(s =>
183 {
184 var type = s.GetInterface(typeof(IValidator<>).Name);
185 if(type != null)
186 {
187 services.AddTransient(type, s);
188 }
189 });
190 }

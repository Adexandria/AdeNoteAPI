33 public async Task<ActionResult> CreateSubThread(CreateThreadDto message, string email, string[] replyEmails, string threadId)
34 {
35 var user = await _userRepository.GetUserByEmail(email);
36
37 if(user == null)
38 {
39 return ActionResult.Failed("Failed create tweet", StatusCodes.Status400BadRequest);
40 }
41
42 var replyUserIds = _userRepository.GetUserIdsByEmails(replyEmails);
43
44 var newThread = new TweetSubThread(threadId, user.Id.ToString(), replyUserIds, message.Message);
45
46 var response = await _chatApplication.CreateSubThread(newThread);
47
48 if (response == null)
49 return ActionResult.Failed("Failed create tweet", StatusCodes.Status400BadRequest);
50
51 var cachedThread = _cacheService.Get<ThreadDtos>($"{_threadCacheKey}:{threadId}");
52
53 if(cachedThread == null)
54 {
55 cachedThread = await _chatApplication.FetchThread(threadId);
56 }
57 else
58 {
59 if (cachedThread.SubThreads == null)
60 {
61 cachedThread.SubThreads = new List<SubThreadDtos>() { response };
62 }
63 else
64 {
65 cachedThread.SubThreads.Add(response);
66 }
67
68 }
69
70 _cacheService.Set($"{_threadCacheKey}:{response.ThreadId}", cachedThread);
71
72 var threadOwners = _userRepository.GetFullNamesById(cachedThread.UserIds);
73
74 var emails = threadOwners.Select(s => new Email(s.Email, "Someone replied to your tweet")).ToList();
75
76 var substitutions = threadOwners.Select(s => new Dictionary<string, string>
77 {
78 { "[Recipient's Name]", s.Name},
79 {"[Original Tweet Content]",cachedThread.Message },
80 { "[Replier's Name]",user.Email},
81 {"[Reply Content]", message.Message},
82 {"[Tweet URL]" , $"{_clientUrl}api/v1/chats/threads/{cachedThread.Id}" },
83 }).ToList();
84
85 _notificationService.SendNotification(emails, EmailTemplate.TweetNotification, ContentType.html, substitutions);
86
87 return ActionResult.SuccessfulOperation();
88 }
